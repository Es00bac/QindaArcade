cmake_minimum_required(VERSION 3.20)
project(QindaArcade VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.24)
pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
find_package(Threads REQUIRED)
find_library(GL_LIBRARY NAMES GL REQUIRED)
option(QINDA_SANITIZERS "Enable address and undefined behavior checks" OFF)
add_executable(qinda-livery-export tools/export_liveries.cpp)
target_include_directories(qinda-livery-export PRIVATE games/common)
target_link_libraries(qinda-livery-export PRIVATE PkgConfig::CAIRO)
add_custom_command(OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/assets/textures/racer-liveries-v2.png"
  COMMAND qinda-livery-export "${CMAKE_CURRENT_SOURCE_DIR}/assets/textures"
  DEPENDS qinda-livery-export games/common/livery.hpp games/common/racer_profiles.hpp
  COMMENT "Exporting editable character liveries and UV island guides")
add_custom_target(art-liveries ALL DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/textures/racer-liveries-v2.png")
foreach(game brawl circuit)
  set(root "${CMAKE_CURRENT_SOURCE_DIR}/games/${game}")
  foreach(shader mesh.vert mesh.frag full.vert shadow.frag sky.frag blur.frag post.frag)
    string(REPLACE "." "_" key "${shader}")
    string(TOUPPER "${key}" key)
    file(READ "${root}/shaders/${shader}" "${key}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/shaders/${shader}")
  endforeach()
  configure_file("${root}/shaders/shader_strings.hpp.in" "${game}/shader_strings.hpp" @ONLY)
  if(game STREQUAL "brawl")
    set(simulation battle)
    set(binary qinda-mega-brawl)
  else()
    set(simulation race)
    set(binary qinda-prism-kart)
  endif()
  add_library(${game}-scene "${root}/src/meshes.cpp" "${root}/src/models.cpp" "${root}/src/${simulation}.cpp" "${root}/src/director.cpp" "${root}/src/export.cpp")
  if(game STREQUAL "circuit")
    target_sources(${game}-scene PRIVATE "${root}/src/track.cpp" "${root}/src/landscape.cpp")
  else()
    target_sources(${game}-scene PRIVATE "${root}/src/arenas.cpp")
  endif()
  target_include_directories(${game}-scene PUBLIC "${root}/include" games/common)
  target_compile_definitions(${game}-scene PUBLIC QINDA_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  add_executable(${binary} "${root}/src/main.cpp" "${root}/src/renderer.cpp")
  add_dependencies(${binary} art-liveries)
  target_include_directories(${binary} PRIVATE "${root}/compat" "${CMAKE_CURRENT_BINARY_DIR}/${game}")
  target_link_libraries(${binary} PRIVATE ${game}-scene PkgConfig::SDL2 PkgConfig::CAIRO "${GL_LIBRARY}" Threads::Threads)
  target_compile_definitions(${binary} PRIVATE BRAWL_SYSTEM_SDL=1 BRAWL_SYSTEM_CAIRO=1 STARWARD_SYSTEM_SDL=1 STARWARD_SYSTEM_CAIRO=1)
  foreach(target ${game}-scene ${binary})
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(QINDA_SANITIZERS)
      target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
  endforeach()
  if(BUILD_TESTING)
    add_executable(${game}-tests "${root}/tests/tests.cpp")
    target_link_libraries(${game}-tests PRIVATE ${game}-scene)
    add_test(NAME ${game}-simulation COMMAND ${game}-tests)
    add_test(NAME ${game}-help COMMAND ${binary} --help)
    add_executable(${game}-player-tests "${root}/tests/player_tests.cpp")
    target_link_libraries(${game}-player-tests PRIVATE ${game}-scene)
    add_test(NAME ${game}-player-input COMMAND ${game}-player-tests)
  endif()
  install(TARGETS ${binary} RUNTIME DESTINATION bin)
endforeach()
install(DIRECTORY assets/ DESTINATION share/qinda-arcade)
if(BUILD_TESTING)
  add_executable(circuit-art-tests games/circuit/tests/art_tests.cpp)
  target_link_libraries(circuit-art-tests PRIVATE circuit-scene)
  add_test(NAME circuit-authored-art COMMAND circuit-art-tests)
  add_executable(controller-tests games/common/tests/controller_tests.cpp)
  target_include_directories(controller-tests PRIVATE games/common)
  target_link_libraries(controller-tests PRIVATE PkgConfig::SDL2)
  target_compile_definitions(controller-tests PRIVATE QINDA_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  if(QINDA_SANITIZERS)
    target_compile_options(controller-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(controller-tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME controller-hotplug-mapping COMMAND controller-tests)
  set_tests_properties(controller-hotplug-mapping PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;QINDA_CONFIG=${CMAKE_CURRENT_BINARY_DIR}/test-config")
endif()
